// grid[r][c] === 1 is an obstacle
function uniquePaths(grid) {
  const m = grid.length, n = grid[0].length;
  const dp = Array.from({ length: m }, () => Array(n).fill(0));  // paths to (r, c)
  for (let r = 0; r < m; r++) {
    for (let c = 0; c < n; c++) {
      if (grid[r][c] === 1) {
        dp[r][c] = 0;                           // @mark blocked
      } else if (r === 0 && c === 0) {
        dp[r][c] = 1;                           // @mark start
      } else {
        const up = r > 0 ? dp[r - 1][c] : 0;
        const left = c > 0 ? dp[r][c - 1] : 0;
        dp[r][c] = up + left;                   // @mark fill
      }
    }
  }
  return dp[m - 1][n - 1];                      // @mark done
}
