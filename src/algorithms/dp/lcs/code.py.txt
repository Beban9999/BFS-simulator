def lcs(a, b):
    m, n = len(a), len(b)
    dp = [[0] * (n + 1) for _ in range(m + 1)]  # @mark init
    # dp[i][j] = LCS length of a[:i] and b[:j]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1     # @mark match
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])  # @mark skip
    return dp[m][n]                             # @mark done

def lcs_string(a, b, dp):                       # walk back from the corner
    i, j, out = len(a), len(b), []
    while i > 0 and j > 0:
        if a[i - 1] == b[j - 1]:
            out.append(a[i - 1])                # @mark take
            i, j = i - 1, j - 1
        elif dp[i - 1][j] >= dp[i][j - 1]:
            i -= 1                              # @mark up
        else:
            j -= 1                              # @mark left
    return ''.join(reversed(out))
