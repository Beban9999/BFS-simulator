def knapsack(weights, values, W):
    n = len(weights)
    # dp[i][w] = best value using only the first i items with capacity w
    dp = [[0] * (W + 1) for _ in range(n + 1)]  # @mark init
    for i in range(1, n + 1):
        wt, val = weights[i - 1], values[i - 1]
        for w in range(W + 1):
            dp[i][w] = dp[i - 1][w]             # @mark skip
            if wt <= w and dp[i - 1][w - wt] + val > dp[i][w]:
                dp[i][w] = dp[i - 1][w - wt] + val  # @mark take
    return dp[n][W]                             # @mark done

def chosen(weights, dp, W):                     # walk back from dp[n][W]
    items, w = [], W
    for i in range(len(weights), 0, -1):
        if dp[i][w] != dp[i - 1][w]:            # @mark pick
            items.append(i - 1)
            w -= weights[i - 1]
    return items[::-1]

# Same answer in O(W) memory: go DOWN in w so each item is counted once
def knapsack_1d(weights, values, W):
    dp = [0] * (W + 1)
    for wt, val in zip(weights, values):
        for w in range(W, wt - 1, -1):
            dp[w] = max(dp[w], dp[w - wt] + val)
    return dp[W]
