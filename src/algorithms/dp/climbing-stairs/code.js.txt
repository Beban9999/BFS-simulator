// ways(n) = ways(n-1) + ways(n-2): the last move was 1 step or 2 steps

// 1) Plain recursion — exponential: the same n is solved again and again
function climbRec(n) {
  if (n <= 1) return 1;                         // @mark r-base
  return climbRec(n - 1) + climbRec(n - 2);     // @mark r-call
}

// 2) Memoisation (top-down) — each n is solved once, then looked up
function climbMemo(n, memo = new Map()) {
  if (n <= 1) return 1;                         // @mark m-base
  if (memo.has(n)) return memo.get(n);          // @mark m-hit
  memo.set(n, climbMemo(n - 1, memo) + climbMemo(n - 2, memo));  // @mark m-store
  return memo.get(n);
}

// 3) Tabulation (bottom-up) — fill the table from the base cases up
function climbTable(n) {
  const dp = Array(n + 1).fill(1);              // @mark t-init
  for (let i = 2; i <= n; i++) {
    dp[i] = dp[i - 1] + dp[i - 2];              // @mark t-fill
  }
  return dp[n];                                 // @mark t-done
}

// 4) Two variables — only the last two entries are ever read
function climbVars(n) {
  let a = 1, b = 1;                             // @mark v-init
  for (let i = 2; i <= n; i++) {
    [a, b] = [b, a + b];                        // @mark v-step
  }
  return b;                                     // @mark v-done
}
