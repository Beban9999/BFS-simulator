def greedy(coins, amount):
    count = 0
    for c in sorted(coins, reverse=True):       # @mark greedy
        count += amount // c                    # take all that fit
        amount %= c
    return count if amount == 0 else -1         # may be wrong!

def fewest(coins, amount):
    dp = [0] + [float("inf")] * amount
    for a in range(1, amount + 1):              # @mark dp
        for c in coins:                         # try each last coin
            if c <= a:
                dp[a] = min(dp[a], dp[a - c] + 1)
    return dp[amount] if dp[amount] != float("inf") else -1
