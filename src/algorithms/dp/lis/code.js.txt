// O(n²): dp[i] = length of the longest increasing subsequence ending at i
function lisQuadratic(nums) {
  const n = nums.length;
  const dp = Array(n).fill(1), prev = Array(n).fill(-1);  // @mark q-init
  for (let i = 0; i < n; i++) {
    for (let j = 0; j < i; j++) {
      if (nums[j] < nums[i] && dp[j] + 1 > dp[i]) {       // @mark q-try
        dp[i] = dp[j] + 1; prev[i] = j;                    // @mark q-take
      }
    }
  }
  let end = 0;
  for (let i = 1; i < n; i++) if (dp[i] > dp[end]) end = i;  // @mark q-best
  const seq = [];
  for (; end !== -1; end = prev[end]) seq.push(nums[end]);  // @mark q-trace
  return seq.reverse();
}

// O(n log n): tails[k] = smallest possible tail of an increasing subsequence of length k+1
function lisLength(nums) {
  const tails = [];                             // @mark b-init
  for (const x of nums) {
    let lo = 0, hi = tails.length;              // @mark b-search
    while (lo < hi) {                           // first tail >= x
      const mid = (lo + hi) >> 1;
      if (tails[mid] < x) lo = mid + 1; else hi = mid;
    }
    if (lo === tails.length) tails.push(x);     // @mark b-append
    else tails[lo] = x;                         // @mark b-replace
  }
  return tails.length;                          // @mark b-done
}
