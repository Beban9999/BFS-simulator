function rob(nums) {
  const n = nums.length;
  if (n === 0) return [0, []];                  // @mark empty
  const dp = Array(n).fill(0);                  // dp[i] = best loot from houses 0..i
  dp[0] = nums[0];                              // @mark base0
  if (n > 1) dp[1] = Math.max(nums[0], nums[1]);  // @mark base1
  for (let i = 2; i < n; i++) {
    dp[i] = Math.max(dp[i - 1], dp[i - 2] + nums[i]);  // @mark fill
  }
  return [dp[n - 1], houses(nums, dp)];         // @mark done
}

// walk back to see what was robbed
function houses(nums, dp) {
  let i = nums.length - 1;
  const picked = [];
  while (i >= 0) {
    if (i > 0 && dp[i] === dp[i - 1]) {
      i -= 1;                                   // @mark skip
    } else {
      picked.push(i);                           // @mark pick
      i -= 2;
    }
  }
  return picked.reverse();
}
