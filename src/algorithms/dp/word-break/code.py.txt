def word_break(s, word_dict):
    words = set(word_dict)
    n = len(s)
    dp = [True] + [False] * n                   # @mark init
    # dp[i]: the prefix s[:i] can be split into dictionary words
    for i in range(1, n + 1):
        for j in range(i):                      # last word is s[j:i]
            if dp[j] and s[j:i] in words:       # @mark check
                dp[i] = True                    # @mark set
                break
    return dp[n]                                # @mark done
