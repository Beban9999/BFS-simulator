def min_distance(a, b):                         # turn a into b
    m, n = len(a), len(b)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(m + 1):
        dp[i][0] = i                            # @mark base-col
    for j in range(n + 1):
        dp[0][j] = j                            # @mark base-row
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if a[i - 1] == b[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]     # @mark same
            else:
                dp[i][j] = 1 + min(dp[i - 1][j - 1],   # replace   # @mark edit
                                   dp[i - 1][j],       # delete a[i-1]
                                   dp[i][j - 1])       # insert b[j-1]
    return dp[m][n]                             # @mark done

def alignment(a, b, dp):                        # walk back, newest operation first
    i, j, ops = len(a), len(b), []
    while i > 0 or j > 0:                       # @mark align
        if i > 0 and j > 0 and a[i - 1] == b[j - 1] and dp[i][j] == dp[i - 1][j - 1]:
            ops.append(('=', a[i - 1], b[j - 1])); i, j = i - 1, j - 1
        elif i > 0 and j > 0 and dp[i][j] == dp[i - 1][j - 1] + 1:
            ops.append(('R', a[i - 1], b[j - 1])); i, j = i - 1, j - 1
        elif i > 0 and dp[i][j] == dp[i - 1][j] + 1:
            ops.append(('D', a[i - 1], '-')); i -= 1
        else:
            ops.append(('I', '-', b[j - 1])); j -= 1
    return ops[::-1]
