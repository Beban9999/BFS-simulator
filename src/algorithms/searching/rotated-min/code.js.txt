function findMin(nums) {
  let lo = 0, hi = nums.length - 1;                 // @mark init
  while (lo < hi) {                                 // @mark loop
    const mid = (lo + hi) >> 1;                     // @mark mid
    if (nums[mid] > nums[hi]) {                     // @mark compare
      lo = mid + 1;                                 // the drop is right of mid   // @mark go-right
    } else {
      hi = mid;                                     // mid could be the minimum   // @mark go-left
    }
  }
  return nums[lo];                                  // @mark done
}
