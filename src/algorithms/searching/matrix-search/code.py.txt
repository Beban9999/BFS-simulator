def search_matrix(matrix, target):
    R, C = len(matrix), len(matrix[0])
    lo, hi = 0, R * C - 1                       # virtual sorted array of R*C cells   # @mark init
    while lo <= hi:                             # @mark loop
        mid = (lo + hi) // 2
        r, c = divmod(mid, C)                   # index -> (row, col)   # @mark map
        if matrix[r][c] == target:              # @mark compare
            return True                         # @mark found
        if matrix[r][c] < target:
            lo = mid + 1                        # @mark go-right
        else:
            hi = mid - 1                        # @mark go-left
    return False                                # @mark none
