def search_matrix(matrix, target):
    r, c = 0, len(matrix[0]) - 1                # top-right corner   # @mark init
    while r < len(matrix) and c >= 0:           # @mark loop
        value = matrix[r][c]                    # @mark look
        if value == target:
            return True                         # @mark found
        if value > target:
            c -= 1                              # whole column below is bigger too   # @mark left
        else:
            r += 1                              # whole row to the left is smaller too   # @mark down
    return False                                # @mark none
