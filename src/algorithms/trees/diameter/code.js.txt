function diameterOfBinaryTree(root) {
  let best = 0;                                 // longest path seen, in edges
  function height(node) {                       // @mark call
    if (node === null) return 0;                // @mark base
    const left = height(node.left);             // @mark left
    const right = height(node.right);           // @mark right
    best = Math.max(best, left + right);        // longest path that bends at this node   // @mark update
    return 1 + Math.max(left, right);           // only one side can continue upward   // @mark combine
  }
  height(root);
  return best;                                  // @mark done
}
