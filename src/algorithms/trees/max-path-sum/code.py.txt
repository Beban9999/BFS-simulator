def max_path_sum(root):
    best = float("-inf")
    def gain(node):                             # best sum of a path going DOWN from node   # @mark call
        nonlocal best
        if node is None:                        # @mark base
            return 0
        left = max(gain(node.left), 0)          # a negative arm is better left out   # @mark left
        right = max(gain(node.right), 0)        # @mark right
        best = max(best, node.val + left + right)   # the path that bends here   # @mark update
        return node.val + max(left, right)      # going up, only one arm can come along   # @mark combine
    gain(root)
    return best                                 # @mark done
