def is_balanced(root):
    def height(node):                           # height, or -1 if this subtree is unbalanced   # @mark call
        if node is None:                        # @mark base
            return 0
        left = height(node.left)                # @mark left
        if left == -1:
            return -1                           # already unbalanced below: stop early   # @mark left-bad
        right = height(node.right)              # @mark right
        if right == -1 or abs(left - right) > 1:   # @mark check
            return -1                           # @mark bad
        return 1 + max(left, right)             # @mark combine
    return height(root) != -1                   # @mark done
