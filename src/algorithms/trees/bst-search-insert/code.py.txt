def search_bst(root, target):
    node = root                                 # @mark s-init
    while node:
        if target == node.val:                  # @mark s-compare
            return node                         # @mark s-found
        node = node.left if target < node.val else node.right   # discard the other half   # @mark s-move
    return None                                 # fell off the tree: not present   # @mark s-missing

def insert_bst(root, val):
    if root is None:
        return TreeNode(val)                    # @mark i-empty
    node = root                                 # @mark i-init
    while True:
        if val < node.val:                      # @mark i-compare
            if node.left is None:
                node.left = TreeNode(val)       # found the empty spot   # @mark i-attach-left
                return root
            node = node.left                    # @mark i-left
        else:
            if node.right is None:
                node.right = TreeNode(val)      # @mark i-attach-right
                return root
            node = node.right                   # @mark i-right
