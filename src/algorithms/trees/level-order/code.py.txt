from collections import deque

def level_order(root, zigzag=False):
    res, queue = [], deque([root] if root else [])   # @mark init
    while queue:
        level = []
        for _ in range(len(queue)):             # exactly the nodes of this level   # @mark level
            node = queue.popleft()              # @mark pop
            level.append(node.val)
            if node.left: queue.append(node.left)   # children wait for the next level   # @mark push
            if node.right: queue.append(node.right)
        if zigzag and len(res) % 2 == 1:
            level.reverse()                     # every other level right-to-left   # @mark zigzag
        res.append(level)                       # @mark add-level
    return res                                  # @mark done
