def morris_inorder(root):
    out, cur = [], root                         # @mark init
    while cur:
        if cur.left is None:
            out.append(cur.val)                 # nothing on the left: visit now   # @mark visit
            cur = cur.right                     # (this may follow a thread back up)   # @mark go-right
        else:
            pred = cur.left                     # find cur's inorder predecessor:
            while pred.right and pred.right is not cur:
                pred = pred.right               # the rightmost node of the left subtree   # @mark find-pred
            if pred.right is None:
                pred.right = cur                # first visit: leave a thread back to cur   # @mark thread
                cur = cur.left                  # @mark go-left
            else:
                pred.right = None               # second visit: the left side is done, remove the thread   # @mark unthread
                out.append(cur.val)             # @mark visit-after
                cur = cur.right                 # @mark leave
    return out                                  # @mark done
