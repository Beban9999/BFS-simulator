class FenwickTree {                             // binary indexed tree, 1-indexed inside
  constructor(nums) {
    this.n = nums.length;
    this.tree = new Array(this.n + 1).fill(0);  // tree[i] = sum of nums over (i - lowbit(i), i]
    nums.forEach((x, i) => this.add(i, x));     // @mark build
  }

  add(i, delta) {                               // nums[i] += delta
    for (i += 1; i <= this.n; i += i & -i) {    // jump to the next node covering i
      this.tree[i] += delta;                    // every node whose range contains i   // @mark add-step
    }
  }

  prefix(i) {                                   // sum of nums[0..i]
    let s = 0;
    for (i += 1; i > 0; i -= i & -i) {          // drop the lowest set bit: next disjoint block
      s += this.tree[i];                        // @mark sum-step
    }
    return s;
  }

  rangeSum(l, r) {
    return this.prefix(r) - (l > 0 ? this.prefix(l - 1) : 0);   // @mark range
  }
}
