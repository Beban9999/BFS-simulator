def range_add(n, updates):
    diff = [0] * (n + 1)                    # @mark init
    for l, r, v in updates:                 # @mark update-loop
        diff[l] += v                        # @mark mark-start
        diff[r + 1] -= v                    # @mark mark-end
    result, running = [0] * n, 0
    for i in range(n):                      # @mark sweep-loop
        running += diff[i]                  # @mark sweep
        result[i] = running
    return result                           # @mark done
