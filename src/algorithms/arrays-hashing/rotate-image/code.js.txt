function rotate(matrix) {
  const n = matrix.length;
  // 1) transpose: swap across the main diagonal
  for (let r = 0; r < n; r++) {                         // @mark transpose-loop
    for (let c = r + 1; c < n; c++) {
      [matrix[r][c], matrix[c][r]] = [matrix[c][r], matrix[r][c]];   // @mark transpose
    }
  }
  // 2) reverse every row
  for (const row of matrix) {                           // @mark reverse-loop
    for (let left = 0, right = n - 1; left < right; left++, right--) {
      [row[left], row[right]] = [row[right], row[left]];  // @mark reverse
    }
  }
  return matrix;                                        // @mark done
}
