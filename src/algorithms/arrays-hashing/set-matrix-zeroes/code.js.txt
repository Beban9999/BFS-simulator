function setZeroes(matrix) {
  const R = matrix.length, C = matrix[0].length;
  const row0 = matrix[0].some((v) => v === 0);        // @mark scan-first
  const col0 = matrix.some((row) => row[0] === 0);
  // 1) record zeros in the first row / column
  for (let r = 1; r < R; r++) {                       // @mark mark-loop
    for (let c = 1; c < C; c++) {
      if (matrix[r][c] === 0) {                       // @mark check
        matrix[r][0] = matrix[0][c] = 0;              // @mark mark
      }
    }
  }
  // 2) zero every cell whose row or column is marked
  for (let r = 1; r < R; r++) {                       // @mark fill-loop
    for (let c = 1; c < C; c++) {
      if (matrix[r][0] === 0 || matrix[0][c] === 0) { // @mark fill-check
        matrix[r][c] = 0;                             // @mark fill
      }
    }
  }
  // 3) finally the first row / column themselves
  if (row0) {                                         // @mark first-row
    for (let c = 0; c < C; c++) matrix[0][c] = 0;
  }
  if (col0) {                                         // @mark first-col
    for (let r = 0; r < R; r++) matrix[r][0] = 0;
  }
  return matrix;                                      // @mark done
}
