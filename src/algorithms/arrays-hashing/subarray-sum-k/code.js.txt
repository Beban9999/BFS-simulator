function subarraySum(nums, k) {
  const count = new Map([[0, 1]]);  // prefix total -> times seen   // @mark init
  let total = 0, answer = 0;
  for (const x of nums) {                               // @mark loop
    total += x;                                         // @mark add
    answer += count.get(total - k) ?? 0;                // @mark lookup
    count.set(total, (count.get(total) ?? 0) + 1);      // @mark store
  }
  return answer;                                        // @mark done
}
