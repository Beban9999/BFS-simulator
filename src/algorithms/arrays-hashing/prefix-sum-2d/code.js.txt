function build(matrix) {
  const R = matrix.length, C = matrix[0].length;
  const P = Array.from({ length: R + 1 }, () => new Array(C + 1).fill(0));  // @mark init
  for (let r = 0; r < R; r++) {                         // @mark build-loop
    for (let c = 0; c < C; c++) {
      P[r + 1][c + 1] = matrix[r][c] + P[r][c + 1]
                      + P[r + 1][c] - P[r][c];          // @mark build
    }
  }
  return P;
}

function regionSum(P, r1, c1, r2, c2) {
  return P[r2 + 1][c2 + 1] - P[r1][c2 + 1]
       - P[r2 + 1][c1] + P[r1][c1];                     // @mark query
}
