function maxSubarray(nums) {
  let best = nums[0], cur = nums[0];            // @mark init
  let start = 0, bestL = 0, bestR = 0;
  for (let i = 1; i < nums.length; i++) {       // @mark loop
    if (cur < 0) {                              // @mark check
      cur = nums[i]; start = i;                 // @mark restart
    } else {
      cur += nums[i];                           // @mark extend
    }
    if (cur > best) {                           // @mark compare
      best = cur; bestL = start; bestR = i;     // @mark update
    }
  }
  return best;                                  // @mark done
}
