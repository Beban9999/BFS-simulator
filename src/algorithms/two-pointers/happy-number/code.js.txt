function nextNumber(n) {
  let total = 0;
  while (n > 0) {
    const digit = n % 10;
    total += digit * digit;
    n = Math.floor(n / 10);
  }
  return total;
}

function isHappy(n) {
  let slow = n, fast = nextNumber(n);                   // @mark init
  while (fast !== 1 && slow !== fast) {                 // @mark loop
    slow = nextNumber(slow);                            // @mark slow-step
    fast = nextNumber(nextNumber(fast));                // @mark fast-step
  }
  return fast === 1;                                    // @mark done
}
