function removeDuplicates(nums) {
  if (nums.length === 0) return 0;
  let write = 1;                                // nums[0..write) is the deduplicated prefix   // @mark init
  for (let read = 1; read < nums.length; read++) {  // @mark loop
    if (nums[read] !== nums[write - 1]) {       // @mark compare
      nums[write] = nums[read];                 // @mark keep
      write++;
    }
  }
  return write;                                 // @mark done
}
