function trap(height) {
  let left = 0, right = height.length - 1;              // @mark init
  let leftMax = 0, rightMax = 0, water = 0;
  while (left < right) {                                // @mark loop
    if (height[left] < height[right]) {                 // @mark compare
      leftMax = Math.max(leftMax, height[left]);        // @mark left-max
      water += leftMax - height[left];                  // @mark left-water
      left++;
    } else {
      rightMax = Math.max(rightMax, height[right]);     // @mark right-max
      water += rightMax - height[right];                // @mark right-water
      right--;
    }
  }
  return water;                                         // @mark done
}
