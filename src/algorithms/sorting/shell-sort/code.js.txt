function shellSort(nums) {
  for (let gap = nums.length >> 1; gap > 0; gap >>= 1) {   // n/2, n/4, ..., 1   // @mark gap
    for (let i = gap; i < nums.length; i++) {   // gap-insertion sort   // @mark pass
      const key = nums[i];
      let j = i;
      while (j >= gap && nums[j - gap] > key) { // @mark compare
        nums[j] = nums[j - gap];                // shift a whole gap to the right   // @mark shift
        j -= gap;
      }
      nums[j] = key;                            // @mark place
    }
  }
  return nums;                                  // @mark done
}
