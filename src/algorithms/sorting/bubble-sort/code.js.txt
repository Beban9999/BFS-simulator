function bubbleSort(nums) {
  const n = nums.length;
  for (let end = n - 1; end > 0; end--) {       // nums[end+1..] is already sorted   // @mark pass
    let swapped = false;
    for (let i = 0; i < end; i++) {             // @mark inner
      if (nums[i] > nums[i + 1]) {              // @mark compare
        [nums[i], nums[i + 1]] = [nums[i + 1], nums[i]];   // @mark swap
        swapped = true;
      }
    }
    if (!swapped) break;                        // no swaps: already sorted   // @mark early-exit
  }
  return nums;                                  // @mark done
}
