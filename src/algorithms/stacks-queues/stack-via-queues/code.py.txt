from collections import deque

class MyStack:
    def __init__(self):
        self.q = deque()                        # front of the queue = top of the stack   # @mark init

    def push(self, x):
        self.q.append(x)                        # @mark push
        for _ in range(len(self.q) - 1):        # rotate the older items behind x
            self.q.append(self.q.popleft())     # @mark rotate

    def pop(self):
        return self.q.popleft()                 # @mark pop

    def top(self):
        return self.q[0]                        # @mark top

    def empty(self):
        return not self.q                       # @mark empty
