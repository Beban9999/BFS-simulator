def car_fleet(target, position, speed):
    cars = sorted(zip(position, speed), reverse=True)   # closest to the target first   # @mark sort
    stack = []                                  # arrival time of each fleet, slowest on top   # @mark init
    for p, s in cars:
        t = (target - p) / s                    # arrival time if this car drove alone   # @mark time
        if stack and t <= stack[-1]:            # catches the fleet ahead before the target   # @mark merge
            continue                            # ...and then moves at that fleet's speed
        stack.append(t)                         # too slow to catch up: a new fleet   # @mark push
    return len(stack)                           # @mark done
