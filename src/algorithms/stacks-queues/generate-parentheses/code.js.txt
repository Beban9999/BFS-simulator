function generateParenthesis(n) {
  const res = [], path = [];                    // path is used like a stack
  function backtrack(opened, closed) {          // @mark call
    if (path.length === 2 * n) {                // @mark full
      res.push(path.join(""));                  // @mark record
      return;
    }
    if (opened < n) {                           // still allowed to open   // @mark try-open
      path.push("(");
      backtrack(opened + 1, closed);
      path.pop();                               // undo the choice   // @mark undo-open
    }
    if (closed < opened) {                      // only close what is open   // @mark try-close
      path.push(")");
      backtrack(opened, closed + 1);
      path.pop();                               // @mark undo-close
    }
  }
  backtrack(0, 0);
  return res;                                   // @mark done
}
