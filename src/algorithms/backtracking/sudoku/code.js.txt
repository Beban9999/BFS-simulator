function solveSudoku(board) {
  const n = board.length;                       // 4 or 9
  const box = n === 4 ? 2 : 3;
  const digits = Array.from({ length: n }, (_, i) => String(i + 1));
  function ok(r, c, d) {
    const br = r - (r % box), bc = c - (c % box);
    for (let k = 0; k < n; k++) {
      if (board[r][k] === d || board[k][c] === d) return false;  // same row or column
      if (board[br + Math.floor(k / box)][bc + (k % box)] === d) return false;  // same box
    }
    return true;
  }
  function solve() {
    for (let r = 0; r < n; r++) {
      for (let c = 0; c < n; c++) {
        if (board[r][c] !== '.') continue;
        for (const d of digits) {               // @mark find
          if (!ok(r, c, d)) continue;           // @mark check
          board[r][c] = d;                      // @mark place
          if (solve()) return true;             // @mark explore
          board[r][c] = '.';                    // @mark undo
        }
        return false;                           // @mark dead
      }
    }
    return true;                                // @mark solved
  }
  return solve();
}
