const KEYS = { 2: 'abc', 3: 'def', 4: 'ghi', 5: 'jkl',
               6: 'mno', 7: 'pqrs', 8: 'tuv', 9: 'wxyz' };

function letterCombinations(digits) {
  if (!digits) return [];                       // @mark empty
  const res = [], path = [];
  function backtrack(i) {
    if (i === digits.length) {                  // @mark full
      res.push(path.join(''));                  // @mark record
      return;
    }
    for (const ch of KEYS[digits[i]]) {         // @mark loop
      path.push(ch);                            // @mark choose
      backtrack(i + 1);                         // @mark explore
      path.pop();                               // @mark unchoose
    }
  }
  backtrack(0);
  return res;                                   // @mark done
}
