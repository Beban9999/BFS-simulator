function pacificAtlantic(heights) {
  const rows = heights.length, cols = heights[0].length;
  // reverse flow: walk uphill from the ocean
  function climb(starts) {
    const seen = new Set(starts.map(([r, c]) => r * cols + c));
    const queue = [...starts];
    for (let head = 0; head < queue.length; head++) {
      const [r, c] = queue[head];                       // @mark pop
      for (const [nr, nc] of [[r + 1, c], [r - 1, c], [r, c + 1], [r, c - 1]]) {
        if (nr >= 0 && nr < rows && nc >= 0 && nc < cols &&
            !seen.has(nr * cols + nc) && heights[nr][nc] >= heights[r][c]) {
          seen.add(nr * cols + nc);                     // @mark climb
          queue.push([nr, nc]);
        }
      }
    }
    return seen;
  }
  const topLeft = [], bottomRight = [];
  for (let c = 0; c < cols; c++) { topLeft.push([0, c]); bottomRight.push([rows - 1, c]); }
  for (let r = 1; r < rows; r++) topLeft.push([r, 0]);
  for (let r = 0; r < rows - 1; r++) bottomRight.push([r, cols - 1]);
  const pacific = climb(topLeft);                       // @mark pacific
  const atlantic = climb(bottomRight);                  // @mark atlantic
  const out = [];
  for (let r = 0; r < rows; r++)
    for (let c = 0; c < cols; c++)
      if (pacific.has(r * cols + c) && atlantic.has(r * cols + c)) out.push([r, c]);
  return out;                                           // @mark done
}
