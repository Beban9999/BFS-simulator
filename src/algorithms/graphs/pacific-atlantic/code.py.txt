from collections import deque

def pacific_atlantic(heights):
    rows, cols = len(heights), len(heights[0])

    def climb(starts):                          # reverse flow: walk uphill from the ocean
        seen = set(starts)
        queue = deque(starts)
        while queue:
            r, c = queue.popleft()              # @mark pop
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if (0 <= nr < rows and 0 <= nc < cols and (nr, nc) not in seen
                        and heights[nr][nc] >= heights[r][c]):
                    seen.add((nr, nc))          # @mark climb
                    queue.append((nr, nc))
        return seen

    top_left = [(0, c) for c in range(cols)] + [(r, 0) for r in range(1, rows)]
    pacific = climb(top_left)                   # @mark pacific
    bottom_right = [(rows - 1, c) for c in range(cols)] + [(r, cols - 1) for r in range(rows - 1)]
    atlantic = climb(bottom_right)              # @mark atlantic
    return sorted(pacific & atlantic)           # @mark done
