function build(n, edges, directed) {
  const adj = Array.from({ length: n }, () => []);              // @mark init
  const matrix = Array.from({ length: n }, () => Array(n).fill(0));
  for (const [u, v] of edges) {                                 // @mark edge
    adj[u].push(v);                                             // @mark add
    matrix[u][v] = 1;
    if (!directed) {
      adj[v].push(u);                                           // @mark add-back
      matrix[v][u] = 1;
    }
  }
  return [adj, matrix];                                         // @mark done
}

// Asking "is there an edge u-v?"
//   matrix: matrix[u][v]          O(1)
//   lists:  adj[u].includes(v)    O(deg(u))
//   edges:  scan the edge list    O(E)
