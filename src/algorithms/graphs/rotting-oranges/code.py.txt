from collections import deque

def oranges_rotting(grid):
    rows, cols = len(grid), len(grid[0])
    queue, fresh = deque(), 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2:
                queue.append((r, c))            # @mark seed
            elif grid[r][c] == 1:
                fresh += 1
    minutes = 0
    while queue and fresh > 0:                  # @mark minute
        minutes += 1
        for _ in range(len(queue)):             # exactly one layer
            r, c = queue.popleft()              # @mark pop
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 1:
                    grid[nr][nc] = 2            # @mark rot
                    fresh -= 1
                    queue.append((nr, nc))
    return minutes if fresh == 0 else -1        # @mark done
