function orangesRotting(grid) {
  const rows = grid.length, cols = grid[0].length;
  let queue = [], fresh = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (grid[r][c] === 2) queue.push([r, c]);   // @mark seed
      else if (grid[r][c] === 1) fresh++;
    }
  }
  let minutes = 0;
  while (queue.length && fresh > 0) {             // @mark minute
    minutes++;
    const next = [];                              // exactly one layer
    for (const [r, c] of queue) {                 // @mark pop
      for (const [dr, dc] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
        const nr = r + dr, nc = c + dc;
        if (nr >= 0 && nr < rows && nc >= 0 && nc < cols && grid[nr][nc] === 1) {
          grid[nr][nc] = 2;                       // @mark rot
          fresh--;
          next.push([nr, nc]);
        }
      }
    }
    queue = next;
  }
  return fresh === 0 ? minutes : -1;              // @mark done
}
