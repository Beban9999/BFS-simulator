def solve(board):
    rows, cols = len(board), len(board[0])

    def mark_safe(r, c):
        if not (0 <= r < rows and 0 <= c < cols) or board[r][c] != 'O':
            return
        board[r][c] = 'S'                       # @mark safe
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            mark_safe(r + dr, c + dc)

    # 1) every 'O' connected to the border can never be surrounded
    for r in range(rows):
        for c in range(cols):
            on_border = r in (0, rows - 1) or c in (0, cols - 1)
            if on_border and board[r][c] == 'O':    # @mark border
                mark_safe(r, c)
    # 2) the remaining 'O's are surrounded: capture them; un-mark the safe ones
    for r in range(rows):
        for c in range(cols):
            if board[r][c] == 'O':
                board[r][c] = 'X'               # @mark capture
            elif board[r][c] == 'S':
                board[r][c] = 'O'               # @mark restore
