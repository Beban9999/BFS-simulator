def num_islands(grid):
    rows, cols = len(grid), len(grid[0])
    seen = set()

    def dfs(r, c):                              # returns the island's area
        if (r < 0 or r >= rows or c < 0 or c >= cols
                or grid[r][c] != '1' or (r, c) in seen):
            return 0
        seen.add((r, c))                        # @mark visit
        return (1 + dfs(r + 1, c) + dfs(r - 1, c)   # @mark recurse
                  + dfs(r, c + 1) + dfs(r, c - 1))

    count = best = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1' and (r, c) not in seen:  # @mark scan
                count += 1
                area = dfs(r, c)                # @mark island-done
                best = max(best, area)
    return count, best                          # @mark done
