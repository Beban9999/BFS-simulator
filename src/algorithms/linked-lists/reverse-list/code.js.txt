function reverseList(head) {
  let prev = null, curr = head;                 // @mark init
  while (curr) {                                // @mark loop
    const nxt = curr.next;                      // save the rest before cutting it off   // @mark save
    curr.next = prev;                           // flip this node's arrow   // @mark flip
    prev = curr;                                // @mark move-prev
    curr = nxt;                                 // @mark move-curr
  }
  return prev;                                  // prev is the new head   // @mark done
}
